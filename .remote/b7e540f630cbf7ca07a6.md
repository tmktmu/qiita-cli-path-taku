```javascript:sample.js
/**
 * 会議日程調整ツール（Google Apps Script）
 *
 * スプレッドシート構成
 *  - 「候補日時」シート … A列に候補日時（例: 11/20(金) 14:00-15:00）、B列に備考
 *  - 「回答」シート     … 名前 / コメント / 更新日時 / 候補日時ごとの ○△× 列
 */

const SHEET_CANDIDATES = '候補日時';
const SHEET_RESPONSES = '回答';
const FIXED_HEADERS = ['名前', 'コメント', '更新日時'];
const MARKS = ['○', '△', '×'];

/** Webアプリの入口 */
function doGet() {
  return HtmlService.createTemplateFromFile('index')
    .evaluate()
    .setTitle('家族会議')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

/** HTMLテンプレートから別ファイル（CSS / JS）を読み込む */
function include(filename) {
  return HtmlService.createHtmlOutputFromFile(filename).getContent();
}

/** スプレッドシートを開く（スプレッドシートにバインドしたスクリプト前提） */
function getSpreadsheet_() {
  return SpreadsheetApp.getActiveSpreadsheet();
}

/**
 * 初期セットアップ（最初に1回だけエディタから実行）
 * シートとヘッダー、サンプルの候補日時を作成します。
 */
function setup() {
  const ss = getSpreadsheet_();

  let cand = ss.getSheetByName(SHEET_CANDIDATES);
  if (!cand) {
    cand = ss.insertSheet(SHEET_CANDIDATES);
    cand.getRange(1, 1, 1, 2).setValues([['候補日時', 'モノ申したいこと']]);
    cand.getRange(2, 1, 3, 2).setValues([
      ['11/20(金) 19:00-20:00', '我が家'],
      ['11/24(火) 20:00-21:00', '鳥貴族'],
      ['11/25(水) 12:00-13:00', '餃子の王将'],
    ]);
    cand.getRange('A:A').setNumberFormat('@'); // 日付の自動変換を防ぐ
    cand.setFrozenRows(1);
    cand.getRange(1, 1, 1, 2).setFontWeight('bold').setBackground('#e8eef7');
    cand.setColumnWidth(1, 220);
  }

  let resp = ss.getSheetByName(SHEET_RESPONSES);
  if (!resp) {
    resp = ss.insertSheet(SHEET_RESPONSES);
    resp.getRange(1, 1, 1, FIXED_HEADERS.length).setValues([FIXED_HEADERS]);
    resp.setFrozenRows(1);
    resp.setFrozenColumns(1);
    resp.getRange(1, 1, 1, FIXED_HEADERS.length).setFontWeight('bold').setBackground('#e8eef7');
  }
}

/** 候補日時を取得 */
function getCandidates_() {
  const sheet = getSpreadsheet_().getSheetByName(SHEET_CANDIDATES);
  if (!sheet || sheet.getLastRow() < 2) return [];
  return sheet
    .getRange(2, 1, sheet.getLastRow() - 1, 2)
    .getDisplayValues()
    .filter(r => String(r[0]).trim() !== '')
    .map(r => ({ label: String(r[0]).trim(), note: String(r[1]).trim() }));
}

/** 画面表示用データを返す（クライアントから呼ばれる） */
function getData() {
  const candidates = getCandidates_();
  const sheet = getSpreadsheet_().getSheetByName(SHEET_RESPONSES);
  const responses = [];

  if (sheet && sheet.getLastRow() >= 2) {
    const values = sheet.getDataRange().getDisplayValues();
    const header = values[0];
    for (let i = 1; i < values.length; i++) {
      const row = values[i];
      if (!row[0]) continue;
      const answers = {};
      candidates.forEach(c => {
        const col = header.indexOf(c.label);
        answers[c.label] = col >= 0 ? row[col] : '';
      });
      responses.push({
        name: row[0],
        comment: row[1],
        updatedAt: row[2],
        answers: answers,
      });
    }
  }
  return { candidates: candidates, responses: responses };
}

/**
 * 回答を保存（同じ名前があれば上書き）
 * @param {{name:string, comment:string, answers:Object<string,string>}} payload
 */
function submitResponse(payload) {
  const name = String(payload && payload.name || '').trim();
  if (!name) throw new Error('名前を入力してください。');
  if (name.length > 50) throw new Error('名前は50文字以内で入力してください。');
  const comment = String(payload.comment || '').trim().slice(0, 500);

  const candidates = getCandidates_();
  const answers = payload.answers || {};
  candidates.forEach(c => {
    if (!MARKS.includes(answers[c.label])) {
      throw new Error('「' + c.label + '」の回答を選択してください。');
    }
  });

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const sheet = getSpreadsheet_().getSheetByName(SHEET_RESPONSES);
    if (!sheet) throw new Error('「回答」シートがありません。setup() を実行してください。');

    // ヘッダーに無い候補日時の列を追加
    let lastCol = Math.max(sheet.getLastColumn(), FIXED_HEADERS.length);
    let header = sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0];
    candidates.forEach(c => {
      if (header.indexOf(c.label) === -1) {
        lastCol++;
        sheet.getRange(1, lastCol).setNumberFormat('@').setValue(c.label)
          .setFontWeight('bold').setBackground('#e8eef7');
        header.push(c.label);
      }
    });

    // 同名の行を探す（無ければ末尾に追加）
    let rowIndex = -1;
    if (sheet.getLastRow() >= 2) {
      const names = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getDisplayValues();
      for (let i = 0; i < names.length; i++) {
        if (names[i][0] === name) { rowIndex = i + 2; break; }
      }
    }
    if (rowIndex === -1) rowIndex = sheet.getLastRow() + 1;

    const now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy/MM/dd HH:mm');
    const rowValues = header.map((h, idx) => {
      if (idx === 0) return name;
      if (idx === 1) return comment;
      if (idx === 2) return now;
      return answers.hasOwnProperty(h) ? answers[h] : '';
    });

    // 既存行の、候補から外れた列の値は消さずに残す
    if (rowIndex <= sheet.getLastRow()) {
      const existing = sheet.getRange(rowIndex, 1, 1, header.length).getValues()[0];
      rowValues.forEach((v, idx) => {
        if (idx >= 3 && !answers.hasOwnProperty(header[idx])) rowValues[idx] = existing[idx];
      });
    }

    const range = sheet.getRange(rowIndex, 1, 1, header.length);
    range.setNumberFormat('@');
    range.setValues([rowValues]);
  } finally {
    lock.releaseLock();
  }
  return getData();
}

/** 回答を削除（名前で指定） */
function deleteResponse(name) {
  name = String(name || '').trim();
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const sheet = getSpreadsheet_().getSheetByName(SHEET_RESPONSES);
    if (sheet && sheet.getLastRow() >= 2) {
      const names = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getDisplayValues();
      for (let i = names.length - 1; i >= 0; i--) {
        if (names[i][0] === name) sheet.deleteRow(i + 2);
      }
    }
  } finally {
    lock.releaseLock();
  }
  return getData();
}
```

```
<script>
  const MARKS = ['○', '△', '×'];
  let state = { candidates: [], responses: [] };

  const $ = (id) => document.getElementById(id);

  function escapeHtml(s) {
    return String(s == null ? '' : s)
      .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
  }

  function showMessage(text, type) {
    const el = $('message');
    el.textContent = text;
    el.className = 'message ' + (type || '');
  }

  function setBusy(busy) {
    $('submitBtn').disabled = busy;
    $('submitBtn').textContent = busy ? '送信中…' : '送信する';
  }

  // ---------- 描画 ----------
  function render(data) {
    state = data;
    renderSummary();
    renderInputs();
  }

  function renderSummary() {
    const { candidates, responses } = state;
    if (!candidates.length) {
      $('summary').innerHTML = '<p class="muted">候補日時がありません。スプレッドシートの「候補日時」シートに追加してください。</p>';
      return;
    }

    // 各候補のスコア（○=2点、△=1点）
    const stats = candidates.map(c => {
      const s = { '○': 0, '△': 0, '×': 0 };
      responses.forEach(r => { if (s.hasOwnProperty(r.answers[c.label])) s[r.answers[c.label]]++; });
      return { ...s, score: s['○'] * 2 + s['△'] };
    });
    const maxScore = Math.max(...stats.map(s => s.score));

    let html = '<table><thead><tr><th>候補日時</th><th>○</th><th>△</th><th>×</th>';
    responses.forEach(r => {
      html += '<th><button type="button" class="name-link" data-name="' + escapeHtml(r.name) + '">'
            + escapeHtml(r.name) + '</button></th>';
    });
    html += '</tr></thead><tbody>';

    candidates.forEach((c, i) => {
      const s = stats[i];
      const best = responses.length > 0 && s.score === maxScore && maxScore > 0;
      html += '<tr class="' + (best ? 'best' : '') + '">'
            + '<th class="cand">' + escapeHtml(c.label)
            + (c.note ? '<span class="cand-note">' + escapeHtml(c.note) + '</span>' : '') + '</th>'
            + '<td class="count mark-○">' + s['○'] + '</td>'
            + '<td class="count mark-△">' + s['△'] + '</td>'
            + '<td class="count mark-×">' + s['×'] + '</td>';
      responses.forEach(r => {
        const m = r.answers[c.label] || '-';
        html += '<td class="mark-' + escapeHtml(m) + '">' + escapeHtml(m) + '</td>';
      });
      html += '</tr>';
    });

    if (responses.length) {
      html += '<tr class="comment-row"><th class="cand">コメント</th><td colspan="3"></td>';
      responses.forEach(r => { html += '<td>' + escapeHtml(r.comment) + '</td>'; });
      html += '</tr>';
    }
    html += '</tbody></table>';
    if (!responses.length) html += '<p class="muted">まだ回答がありません。</p>';
    $('summary').innerHTML = html;

    document.querySelectorAll('.name-link').forEach(btn => {
      btn.addEventListener('click', () => loadForEdit(btn.dataset.name));
    });
  }

  function renderInputs(prefill) {
    const answers = (prefill && prefill.answers) || {};
    let html = '';
    state.candidates.forEach((c, i) => {
      html += '<div class="cand-row"><div class="cand-label">' + escapeHtml(c.label)
            + (c.note ? '<span class="cand-note">' + escapeHtml(c.note) + '</span>' : '')
            + '</div><div class="segmented" role="radiogroup" aria-label="' + escapeHtml(c.label) + '">';
      MARKS.forEach((m, j) => {
        const id = 'c' + i + '_' + j;
        const checked = answers[c.label] === m ? ' checked' : '';
        html += '<input type="radio" name="c' + i + '" id="' + id + '" value="' + m + '"' + checked + '>'
              + '<label for="' + id + '">' + m + '</label>';
      });
      html += '</div></div>';
    });
    $('candidateInputs').innerHTML = html;
  }

  // ---------- 操作 ----------
  function loadForEdit(name) {
    const r = state.responses.find(x => x.name === name);
    if (!r) return;
    $('name').value = r.name;
    $('comment').value = r.comment || '';
    renderInputs(r);
    $('formTitle').textContent = r.name + ' さんの回答を編集';
    $('deleteBtn').classList.remove('hidden');
    showMessage('', '');
    $('answerForm').scrollIntoView({ behavior: 'smooth' });
  }

  function resetForm() {
    $('answerForm').reset();
    renderInputs();
    $('formTitle').textContent = '回答する';
    $('deleteBtn').classList.add('hidden');
    showMessage('', '');
  }

  function collectPayload() {
    const answers = {};
    state.candidates.forEach((c, i) => {
      const checked = document.querySelector('input[name="c' + i + '"]:checked');
      answers[c.label] = checked ? checked.value : '';
    });
    return { name: $('name').value.trim(), comment: $('comment').value.trim(), answers };
  }

  function onError(err) {
    setBusy(false);
    showMessage(err && err.message ? err.message : 'エラーが発生しました。', 'error');
  }

  function load() {
    google.script.run.withSuccessHandler(render).withFailureHandler(onError).getData();
  }

  $('answerForm').addEventListener('submit', (e) => {
    e.preventDefault();
    const payload = collectPayload();
    if (!payload.name) return showMessage('名前を入力してください。', 'error');
    const missing = state.candidates.find(c => !payload.answers[c.label]);
    if (missing) return showMessage('「' + missing.label + '」の回答を選択してください。', 'error');

    setBusy(true);
    google.script.run
      .withSuccessHandler((data) => {
        setBusy(false);
        render(data);
        resetForm();
        showMessage(payload.name + ' さんの回答を保存しました。', 'success');
      })
      .withFailureHandler(onError)
      .submitResponse(payload);
  });

  $('deleteBtn').addEventListener('click', () => {
    const name = $('name').value.trim();
    if (!name) return;
    // ブラウザのconfirmダイアログはGASのiframe内で使用可能
    if (!confirm(name + ' さんの回答を削除しますか？')) return;
    google.script.run
      .withSuccessHandler((data) => { render(data); resetForm(); showMessage('削除しました。', 'success'); })
      .withFailureHandler(onError)
      .deleteResponse(name);
  });

  $('resetBtn').addEventListener('click', resetForm);
  $('reloadBtn').addEventListener('click', load);

  load();
</script>

```
